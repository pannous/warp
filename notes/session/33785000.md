       ($name:ident, $param_type:ty, $kind:expr, box_in: $box_type:expr) => {
           pub fn $name(&mut self) -> u32 {
               let func_type = self.types.len();
               self.types.ty().function(vec![$param_type], vec![Ref(node_ref)]);
               self.functions.function(func_type);

               self.exports.export(stringify!($name), ExportKind::Func, idx);
